    function safeUrl(url){

    function gatewayUrl(value){
      var raw=String(value||"").trim(), gateway=$("ipfsGateway").value.trim()||"https://ipfs.io/ipfs/";
      if(/^ipfs:\/\//i.test(raw)){
        var path=raw.replace(/^ipfs:\/\/(ipfs\/)?/i,"");

      }

        return gateway.replace(/\/+$/,"/")+raw.replace(/^\/+/,"");
      }
      return safeUrl(raw);
    }
    function slug(s){return String(s||"molecule").toLowerCase().replace(/[^a-z0-9]+/g,"-").replace(/^-|-$/g,"").slice(0,70)||"molecule";}

    async function rpc(method,params){
      var url=$("rpcUrl").value.trim();
      if(!/^https?:\/\//i.test(url))throw new Error("Enter an HTTP(S) JSON-RPC URL.");
      var ctl=new AbortController(), timer=setTimeout(function(){ctl.abort();},30000);
      try{
        var res=await fetch(url,{method:"POST",headers:{"content-type":"application/json"},body:JSON.stringify({jsonrpc:"2.0",id:Date.now(),method:method,params:params||[]}),signal:ctl.signal});
        if(!res.ok)throw new Error("RPC HTTP "+res.status);
        var body=await res.json();if(body.error)throw new Error(body.error.message||pretty(body.error));return body.result;
      }finally{clearTimeout(timer);}
    }
    function requireEthers(){
      if(!window.ethers)throw new Error("The ABI helper did not load. Check access to cdn.jsdelivr.net, then reload.");
      return window.ethers;
    }
    function encodeCall(signature,tokenId){
      var E=requireEthers(), selector=E.id(signature).slice(0,10), id=BigInt(tokenId);
      if(id<0n)throw new Error("Token ID must be non-negative.");
      return selector+id.toString(16).padStart(64,"0");
    }
    async function ethCall(signature,tokenId){
      var address=$("contractAddress").value.trim();
      if(!/^0x[0-9a-fA-F]{40}$/.test(address))throw new Error("Enter a valid 20-byte contract address.");
      return rpc("eth_call",[{to:address,data:encodeCall(signature,tokenId)},"latest"]);
    }
    function bytesFromHex(hex){var E=requireEthers();return E.getBytes(hex);}
    function word(bytes,at){
      if(at<0||at+32>bytes.length)throw new Error("ABI word out of bounds.");
      var h="0x";for(var i=at;i<at+32;i++)h+=bytes[i].toString(16).padStart(2,"0");return BigInt(h);
    }
    function decodeSingleString(hex){
      var b=bytesFromHex(hex), off=Number(word(b,0)), len=Number(word(b,off));
      if(!Number.isSafeInteger(len)||off+32+len>b.length)throw new Error("Invalid ABI string.");
      return new TextDecoder().decode(b.slice(off+32,off+32+len));
    }
    function validStringTuple(b,base){
      try{
        var first=Number(word(b,base)), n=first/32;
        if(!Number.isInteger(n)||n<1||n>32)return null;
        var arr=[];
        for(var i=0;i<n;i++){
          var rel=Number(word(b,base+i*32)), pos=base+rel, len=Number(word(b,pos));
          if(!Number.isSafeInteger(len)||pos+32+len>b.length)return null;
          arr.push(new TextDecoder().decode(b.slice(pos+32,pos+32+len)));
        }
        return arr;
      }catch(e){return null;}
    }
    function decodeStringTuple(hex){
      var b=bytesFromHex(hex), top=validStringTuple(b,0), outer=null;
      try{var o=Number(word(b,0));if(o>=32&&o<b.length)outer=validStringTuple(b,o);}catch(e){}
      if(outer&&outer.length>1)return outer;if(top)return top;
      return [decodeSingleString(hex)];
    }
    function decodeBool(hex){return BigInt(hex)!==0n;}
    function decodeAddress(hex){return "0x"+String(hex).replace(/^0x/,"").slice(-40);}
    function addResolvedLink(label,value){
      if(!value)return;
      var href=gatewayUrl(value), shown=String(value);
      linksEl.insertAdjacentHTML("beforeend",'<div class="resolver-link"><span><b>'+esc(label)+':</b> '+(href?'<a href="'+esc(href)+'" target="_blank" rel="noopener">'+esc(shown)+'</a>':esc(shown))+'</span></div>');
    }
    function collectLinks(value,path,out,seen){
      out=out||[];seen=seen||new Set();if(out.length>30)return out;
      if(typeof value==="string"){
        if(/^(ipfs:\/\/|https?:\/\/|ar:\/\/)/i.test(value)&&!seen.has(value)){seen.add(value);out.push({path:path||"link",value:value});}
      }else if(Array.isArray(value)){value.forEach(function(v,i){collectLinks(v,(path||"root")+"["+i+"]",out,seen);});}
      else if(value&&typeof value==="object"){Object.keys(value).forEach(function(k){collectLinks(value[k],path?path+"."+k:k,out,seen);});}
      return out;
    }
    function decodeDataJson(uri){
      if(!/^data:application\/json/i.test(uri))return null;
      var comma=uri.indexOf(","), head=uri.slice(0,comma), data=uri.slice(comma+1);
      var txt=/;base64/i.test(head)?decodeURIComponent(escape(atob(data))):decodeURIComponent(data);
      return JSON.parse(txt);
    }
    function looksBase64(s){
      var x=String(s||"").replace(/\s+/g,"");return x.length>80&&x.length%4===0&&/^[A-Za-z0-9+/]+={0,2}$/.test(x);
    }
    function base64Bytes(s){
      var clean=String(s).replace(/^data:[^,]*;base64,/i,"").replace(/\s+/g,"");
      var approx=Math.floor(clean.length*3/4);if(approx>MAX_AUTO_BYTES)throw new Error("Payload is larger than the 12 MB browser safety limit.");
      var bin=atob(clean), bytes=new Uint8Array(bin.length);for(var i=0;i<bin.length;i++)bytes[i]=bin.charCodeAt(i);return bytes;
    }
    function detectFormat(bytes,name,type){
      var lower=String(name||"").toLowerCase(), ct=String(type||"").toLowerCase();
      if(/\.bcif$/.test(lower)||ct.indexOf("binarycif")>=0)return "bcif";
      if(/\.sdf$|\.mol$/.test(lower))return "sdf";
      if(/\.mol2$/.test(lower))return "mol2";
      if(/\.pdb$|\.ent$/.test(lower))return "pdb";
      if(/\.cif$|\.mmcif$/.test(lower))return "cif";
      if(/\.xyz$/.test(lower))return "xyz";
      var sample=new TextDecoder().decode(bytes.slice(0,Math.min(bytes.length,5000)));
      if(/\$\$\$\$|V2000|V3000/.test(sample))return "sdf";
      if(/^@<TRIPOS>/m.test(sample))return "mol2";
      if(/^(ATOM  |HETATM|HEADER|MODEL )/m.test(sample))return "pdb";
      if(/^data_/m.test(sample)||/_atom_site\./.test(sample))return "cif";
      return "unknown";
    }
    function makeDownload(bytes,name,type){
      var blob=new Blob([bytes],{type:type||"application/octet-stream"}), url=URL.createObjectURL(blob);activeObjectUrls.push(url);
      var a=document.createElement("a");a.className="resolver-download";a.href=url;a.download=name||"molnft-structure.bin";a.textContent="Download retrieved payload";downloadsEl.appendChild(a);
    }
    function renderBytes(bytes,name,type){
      clearUrls();downloadsEl.innerHTML="";makeDownload(bytes,name,type);
      var fmt=detectFormat(bytes,name,type);
      if(fmt==="bcif"){clearViewer("BCIF payload retrieved. Download is available; this lightweight 3Dmol panel does not guarantee BinaryCIF decoding.");return fmt;}
      if(fmt==="unknown"){clearViewer("Payload retrieved, but its molecular format was not recognised. Download is available.");return fmt;}
      try{
        viewerHost.innerHTML="";activeViewer=$3Dmol.createViewer(viewerHost,{backgroundColor:"white"});
        var data=new TextDecoder().decode(bytes);activeViewer.addModel(data,fmt);
        if(fmt==="pdb"||fmt==="cif")activeViewer.setStyle({},{cartoon:{},stick:{radius:0.12}});
        else activeViewer.setStyle({},{stick:{radius:0.14},sphere:{scale:0.28}});
        activeViewer.zoomTo();activeViewer.render();return fmt;
      }catch(e){clearViewer("Payload retrieved, but rendering failed: "+e.message);return fmt;}
    }
    async function fetchResource(input,label){
      var url=gatewayUrl(input);if(!url)throw new Error("Enter a valid IPFS URI/CID or HTTP(S) URL.");
      setStatus("Fetching "+url+" …");
      var res=await fetch(url,{headers:{accept:"application/json,text/plain,chemical/*,application/octet-stream;q=0.9,*/*;q=0.5"}});
      if(!res.ok)throw new Error("Fetch failed: HTTP "+res.status);
      var declared=Number(res.headers.get("content-length")||0);if(declared>MAX_AUTO_BYTES)throw new Error("Remote payload exceeds the 12 MB browser safety limit.");
      var buf=await res.arrayBuffer();if(buf.byteLength>MAX_AUTO_BYTES)throw new Error("Remote payload exceeds the 12 MB browser safety limit.");
      var bytes=new Uint8Array(buf), type=res.headers.get("content-type")||"", name=(new URL(url)).pathname.split("/").pop()||"ipfs-payload";
      if(type.indexOf("json")>=0||/\.json$/i.test(name)){
        var obj=JSON.parse(new TextDecoder().decode(bytes));jsonEl.textContent=pretty(obj);
        collectLinks(obj).forEach(function(x){addResolvedLink(x.path,x.value);});
        setKv({Mode:"IPFS / URL",Source:label||input,Bytes:bytes.length,Type:type||"JSON"});
        var candidates=collectLinks(obj).filter(function(x){return /\.(sdf|mol2?|pdb|ent|cif|mmcif|bcif|xyz)(\?|$)/i.test(x.value)||/(structure|file|animation_url|content)/i.test(x.path);});
        if(candidates[0]){addResolvedLink("Suggested structure",candidates[0].value);clearViewer("Metadata resolved. Use the IPFS/link field to load the suggested structure payload.");}
        else clearViewer("Metadata resolved; no obvious molecular structure link was detected.");
      }else{
        var fmt=renderBytes(bytes,name,type);jsonEl.textContent=pretty({source:input,resolvedURL:url,bytes:bytes.length,contentType:type||null,detectedFormat:fmt});
        setKv({Mode:"IPFS / URL",Source:label||input,Bytes:bytes.length,Format:fmt});
      }
      addResolvedLink("Resolved URL",url);setStatus("Resource resolved successfully.","ok");return {bytes:bytes,type:type,name:name,url:url};
    }
    $("resolveIpfs").addEventListener("click",async function(){

    function stringsToObject(arr,labels){
      var o={};arr.forEach(function(v,i){o[(labels&&labels[i])||("field_"+i)]=v;});return o;
    }
    async function optionalCall(signature,id,decoder){
      try{var raw=await ethCall(signature,id);if(!raw||raw==="0x")return null;return decoder(raw);}catch(e){return null;}
    }
    async function resolveChain(){
      var id=$("liveToken").value.trim();if(id==="")throw new Error("Enter a token or parent ID.");remember();clearOutput();
      setStatus("Connecting to GenesisL1 JSON-RPC …");
      var chainId=await rpc("eth_chainId",[]);
      var address=$("contractAddress").value.trim();
      var exists=await optionalCall("tokenExists(uint256)",id,decodeBool);
      var owner=await optionalCall("ownerOf(uint256)",id,decodeAddress);
      var tokenUri=await optionalCall("tokenURI(uint256)",id,decodeSingleString);
      var record={chainId:chainId,contract:address,tokenId:id,tokenExists:exists,owner:owner,tokenURI:tokenUri};
      setKv({Mode:"GenesisL1 RPC","Chain ID":chainId,Contract:address,"Token ID":id,Exists:exists===null?"method unavailable":String(exists),Owner:owner||"unavailable"});
      if(tokenUri){
        addResolvedLink("tokenURI",tokenUri);
        try{
          var embedded=decodeDataJson(tokenUri);
          if(embedded){record.tokenMetadata=embedded;collectLinks(embedded).forEach(function(x){addResolvedLink("tokenURI."+x.path,x.value);});}
          else if(/^ipfs:\/\/|^https?:\/\//i.test(tokenUri)){
            setStatus("Token found. Resolving tokenURI metadata …");
            try{
              var metaUrl=gatewayUrl(tokenUri), r=await fetch(metaUrl), ct=r.headers.get("content-type")||"";
              if(r.ok&&(ct.indexOf("json")>=0||/\.json(\?|$)/i.test(metaUrl))){record.tokenMetadata=await r.json();collectLinks(record.tokenMetadata).forEach(function(x){addResolvedLink("metadata."+x.path,x.value);});}
            }catch(e){record.tokenUriFetchWarning=e.message;}
          }
        }catch(e){record.tokenUriParseWarning=e.message;}
      }
      if($("fetchPayload").checked){
        setStatus("Basic token data resolved. Requesting the optional full MolNFT payload …");
        var entire=await optionalCall("getEntireNFT(uint256)",id,decodeStringTuple), payload=null, payloadSource="";
        if(entire&&entire.length){
          record.getEntireNFT=stringsToObject(entire,MOL_FIELDS);
          payload=entire[entire.length-1];payloadSource="getEntireNFT";
        }else{
          var combined=await optionalCall("getCombinedData(uint256)",id,decodeSingleString);
          if(combined!=null){record.getCombinedData={characters:combined.length};payload=combined;payloadSource="getCombinedData";}
        }
        if(payload){
          try{
            var bytes;
            if(/^data:[^,]*;base64,/i.test(payload)||looksBase64(payload))bytes=base64Bytes(payload);
            else bytes=new TextEncoder().encode(payload);
            var fmt=renderBytes(bytes,"molnft-"+id+".bcif","application/octet-stream");
            record.payload={source:payloadSource,characters:payload.length,bytes:bytes.length,detectedFormat:fmt};
          }catch(e){record.payload={source:payloadSource,characters:payload.length,notDecoded:e.message};clearViewer(e.message);}
        }else clearViewer("No full payload was returned by getEntireNFT or getCombinedData.");
      }else{
        clearViewer(tokenUri?"Token metadata linked. Enable the full-payload checkbox only when structure reconstruction is required.":"Token queried; no tokenURI was returned.");
      }
      jsonEl.textContent=pretty(record);setStatus("MolNFT read completed. Unavailable contract methods were skipped without failing the whole request.","ok");
    }
    $("resolveChain").addEventListener("click",async function(){
      var b=this;b.disabled=true;try{await resolveChain();}catch(e){setStatus(e.name==="AbortError"?"RPC request timed out.":e.message,"err");clearViewer("MolNFT resolution failed.");}finally{b.disabled=false;}
    });
    $("clearResolver").addEventListener("click",function(){clearOutput();setStatus("Cleared. Configuration fields were preserved.");});
    renderMock();
  })();
